def _rows(value):
    if isinstance(value, list):
        for item in value:
            yield from _rows(item)
    elif isinstance(value, dict):
        if 'scene_id' in value and ('status' in value or 'trajectory' in value):
            yield value
        else:
            for key in ('development', 'archive', 'archive_raw_development', 'scenes', 'results'):
                if key in value:
                    yield from _rows(value[key])


def _diagnose(scene, origin):
    events = (scene.get('trajectory') or {}).get('events', [])
    failures = []
    successes = []
    for event in events:
        if event.get('event') != 'execution_result':
            continue
        if event.get('ok') is False:
            failures.append({
                'step': event.get('step'),
                'stderr': event.get('stderr'),
                'error': event.get('error')
            })
        elif event.get('ok') is True:
            successes.append(event.get('step'))
    missing = scene.get('status') != 'complete' or (scene.get('metrics') or {}).get('valid') is False
    return {
        'scene_id': scene.get('scene_id'),
        'origin': origin,
        'object_id': scene.get('target_object', 'unknown'),
        'observation_type': 'supplied_execution_and_measurement_evidence',
        'observation': ('No valid final output: reconstruction failure.' if missing else
                        'Completed output; inspect supplied reference/render pair for visual disagreement.'),
        'execution_failures': failures,
        'successful_execution_steps': successes,
        'metrics': scene.get('metrics'),

def solve(task, runtime):
    prompt = (Path(__file__).parent / 'system.txt').read_text()
    messages = [{'role': 'system', 'content': prompt}, {'role': 'user', 'content': [
        {'type': 'text', 'text': 'Reconstruct this RGB image. Match its camera and framing. ' +
         f"Image dimensions: {task['width']} x {task['height']}. Budget: {runtime.remaining()}"},
        runtime.image(task['image'])]}]
    while runtime.remaining()['model_calls'] > 0 and runtime.remaining()['seconds'] > 0:
        response = runtime.ask(messages)
        messages.append(response)
        text = response['content'] or ''
        blocks = re.findall(r'```python\s*\n(.*?)```', text, re.S)
        if not blocks:
            if text.strip().upper() == 'DONE': return
            messages.append({'role': 'user', 'content': 'Return one complete program in a python code block, or DONE.'})
            continue
        if runtime.remaining()['executions'] <= 0: return
        result = runtime.execute(blocks[-1])
        feedback = {'type': 'text', 'text': json.dumps({k: v for k, v in result.items() if k != 'image'}) +
                    '\nCompare the rendered camera view with the original reference. ' +
                    f'Fix errors with a complete revised program, or finish with DONE. Remaining: {runtime.remaining()}'}
        content = [feedback]
        if result['image']: content.append(runtime.image(result['image']))
        messages.append({'role': 'user', 'content': content})
