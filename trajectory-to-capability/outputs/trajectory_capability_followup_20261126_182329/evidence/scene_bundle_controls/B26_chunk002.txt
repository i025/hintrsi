"""Evidence-led planning using only the supplied development packet."""


def plan(packet):
    scenes = [s for s in packet.get('development', []) if isinstance(s, dict)]
    observations = packet.get('controlled_probes', []) or []
    used = packet.get('executed_probe_count', len(observations))
    if not isinstance(used, (int, float)):
        used = len(observations)
    available = max(0, 2 - max(int(used), len(observations)))
    seen = set()
    for observation in observations:
        if isinstance(observation, dict):
            seen.add((observation.get('scene_id'), observation.get('probe')))
    failures = 0
    successes = 0
    for scene in scenes:
        trajectory = scene.get('trajectory') or {}
        for event in trajectory.get('events', []):
            if event.get('event') == 'execution_result':
                if event.get('ok') is True:
                    successes += 1
                else:
                    failures += 1
    complete = [s for s in scenes if s.get('status') == 'complete'
                and (s.get('metrics') or {}).get('valid') is True]

    def score(scene):
        value = (scene.get('metrics') or {}).get('overall')
        return value if isinstance(value, (int, float)) else float('inf')

    complete.sort(key=score)
    probes = []
    # Execution starvation is directly observable; further renders cannot diagnose it.
    if failures < max(1, successes):
        for scene in complete:
            if len(probes) >= available:
                break
            sid = scene.get('scene_id')
            if not isinstance(sid, str):
                continue
            functional = scene.get('functional') or {}
            kind = 'isolate' if functional.get('functional_pass') is False else 'camera_focal'
            if (sid, kind) not in seen:
                probes.append({'scene_id': sid, 'kind': kind})
    instruction = (
        'Improve the general shared reconstruction procedure and callable plan(packet), using the '
        'entire supplied development archive, sources, execution traces, images, geometry summaries '
        'and prescribed edit regressions. Do not explore external files or specialize to scene IDs. '
        'First distinguish build failures, missing outputs, exhausted correction opportunities and '
        'visual mismatch. Count actual execution_result events; regression render counts describe '
        'a different operation and must not be substituted for reconstruction successes. '
        'Check whether API safeguards recover first-build success and permit useful visual revisions. '
        'Then assess framing, visible object coverage, silhouette, occlusion and broad tone before '
        'fine detail. Compare all available development cases, including regressions and failures. '
        'Interpret diagnostic interventions only at their actual target and pivot: a tiny component '
        'scale change cannot identify whole-object scale error, and one focal perturbation cannot '
        'establish an optimal camera or true geometry. Missing measurements remain unknown; missing '
        'required output is failure. Structural/edit passes certify only the prescribed operations, '
        'not semantic completeness, physical contact or metric 3D accuracy; AABBs are approximate. '
        'Do not optimize names to manipulate target selection. Preserve meaningful assemblies. '
        'Propose a falsifiable next change, state expected evidence and risks, and retain complete '
        'working replacement files with the fixed APIs, roles, objective, splits and budgets. '
        'Selection aggregates may inform retention only; do not request Selection images or labels. '
        'The latest valid meta program governs the next exploration branch independently of task '
        'program acceptance. No independent evaluation or rendering is permitted in the proposer session.'
    )
    return {'probes': probes, 'regression_guidance': True, 'instruction': instruction}

from pathlib import Path


def solve(task, runtime):
    prompt = (Path(__file__).parent / 'system.txt').read_text()
    messages = [
        {'role': 'system', 'content': prompt},
        {'role': 'user', 'content': [
            {'type': 'text', 'text': (
                'Reconstruct this RGB image. Match camera, framing, occlusions and major color regions. '
                f"Image dimensions: {task['width']} x {task['height']}. "
                f'Budget: {runtime.remaining()}. '
                'First produce a complete economical scene with all major visible objects. '
                'In brief code comments record approximate normalized image landmarks, camera assumptions, '
                'and the principal uncertain proportions. Audit API constraints before returning code. '
                'Keep camera, object dimensions and lighting parameters easy to revise. '
                'Do not spend the first build on dense subpixel decoration.'
            )},
            runtime.image(task['image'])
        ]}
    ]
    successful = []
    calls = 0
    executions = 0
    while calls < 5:
        budget = runtime.remaining()
        if budget['model_calls'] <= 0 or budget['seconds'] <= 0:
            break
        can_build = executions < 3 and budget['executions'] > 0
        if not can_build and not successful:
            break
        response = runtime.ask(messages)
        calls += 1
        messages.append(response)
        text = response.get('content') or ''
        if not isinstance(text, str):
            text = ''
        selection = re.fullmatch(r'SELECT\s+(\d+)\s*', text.strip(), re.I)
        if selection:
            step = int(selection.group(1))
            if step in successful:
                runtime.select(step)
                return
        if text.strip().upper() == 'DONE' and successful:
            runtime.select(successful[-1])
            return
        blocks = re.findall(r'```python\s*\n(.*?)```', text, re.S)
        if not can_build:
            messages.append({'role': 'user', 'content': (
                f'No executions remain. Select an observed successful build: {successful}. '
                'Reply SELECT followed by its step number. Do not generate unexecutable revisions.'
            )})
            continue
        if len(blocks) != 1:
            messages.append({'role': 'user', 'content': (
                'Return exactly one complete Python program in one python code block. '
                f'Alternatively SELECT an observed successful step from {successful}, or DONE if one exists. '
                'No successful output means reconstruction failure, not completion.'
            )})
            continue
        budget = runtime.remaining()
        if budget['executions'] <= 0 or budget['seconds'] <= 0:
            break
        result = runtime.execute(blocks[0])
        executions += 1
        visible = bool(result.get('ok') and result.get('image'))
        step = result.get('step')
        if visible and isinstance(step, int):
            successful.append(step)
        report = {k: v for k, v in result.items() if k != 'image'}
        remaining = runtime.remaining()
        can_retry = executions < 3 and remaining['executions'] > 0
        if not visible:
            guidance = (
                'BUILD/OUTPUT FAILURE: no successful rendered view is available for this attempt. '
                'Do not claim a visual comparison. Read the error and audit the whole program for '
                'related faults, especially fillet radii, missing World, repeated include keys, '
                'zero-length rods and operations performed before realize. '
                'Return a complete corrected program if executions remain; preserve the intended composition.'
            )
        elif can_retry:
            guidance = (
                'Compare this render with the original reference in the first user message. '
                'In brief comments identify the three largest visible discrepancies and the parameter '
                'changes intended to fix them. Check image borders, major object extents, floor/wall '
                'boundaries, edge slopes, foreground/background size ratios, occlusions and broad tones. '
                'If errors move coherently across the scene, correct camera orientation/framing; '
                'if localized, correct the relevant object or placement. Do not change focal length '
                'and object scale indiscriminately. Preserve already matching regions and editable groups. '
                'Return one complete revised program, or SELECT an existing successful step if further '
                'changes are unjustified. Add fine detail only after composition is credible.'
            )
        else:
            guidance = (
                'No executions remain. Compare the successful rendered builds already shown with the '
                'reference. Reply SELECT followed by the best successful step number. '
                'Prefer framing, silhouettes, visibility and broad colors over detail count. '
                'Do not return a revised program.'
            )
        content = [{'type': 'text', 'text': (
            json.dumps(report, default=str) + '\n' + guidance +
            f'\nSuccessful steps: {successful}. Remaining: {remaining}'
        )}]
        if visible:
            content.append(runtime.image(result['image']))
        messages.append({'role': 'user', 'content': content})
    if successful:
        runtime.select(successful[-1])

def _dict(value):
    return value if isinstance(value, dict) else {}


def _list(value):
    return value if isinstance(value, list) else []


def _number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool) and math.isfinite(value)


def _scenes(value):
    # Called only on explicitly supplied raw development evidence, never Selection.
    if isinstance(value, list):
        for item in value:
            yield from _scenes(item)
    elif isinstance(value, dict):
        if 'scene_id' in value and any(k in value for k in ('trajectory', 'status', 'metrics')):
            yield value
        else:
            for key in ('development', 'archive_raw_development', 'scenes'):
                if key in value:
                    yield from _scenes(value[key])


def _execution_summary(scene):
    events = [e for e in _list(_dict(scene.get('trajectory')).get('events'))
              if isinstance(e, dict)]
    results = [e for e in events if e.get('event') == 'execution_result']
    ok = [e for e in results if e.get('ok') is True]
    failed = [e for e in results if e.get('ok') is False]
    unknown = [e for e in results if e.get('ok') is not True and e.get('ok') is not False]
    # Lossless projected traces can omit the image key; absence alone is unknown.
    explicit_missing = [e for e in ok if 'image' in e and not e['image']]

    functional = _dict(scene.get('functional'))

        'explicit_missing_render_events': len(explicit_missing),
        'timeout_events': sum('timeout' in str(e.get('error', '')).lower() for e in failed),
        'first_execution_ok': results[0].get('ok') if results else None,

def _location(program, target):
    # Only an exact literal include slot is a proven static mapping.
    # Dynamic names, helper expansion and Blender suffixes remain unknown.
    if not isinstance(target, str) or target == 'unknown':
        return 'unknown'
    try:
        tree = ast.parse(program)
    except (SyntaxError, TypeError):
        return 'unknown'
    lines = []
    for node in ast.walk(tree):
        if (isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute)
                and node.func.attr == 'include' and node.args
                and isinstance(node.args[0], ast.Constant)
                and node.args[0].value == target):
            lines.append(node.lineno)
    return 'program:line ' + str(lines[0]) if len(lines) == 1 else 'unknown'


def _dependencies(scene, target):
    rows = scene.get('structure', {}).get('objects', [])
    by_name = {r.get('name'): r for r in rows}
    regression = scene.get('regression', {})
    selection = regression.get('target_selection') or scene.get('structure', {}).get('target', {})
    meshes = selection.get('mesh_names', [])
    materials = sorted({m for name in meshes for m in by_name.get(name, {}).get('material_names', [])})
    parent = by_name.get(target, {}).get('parent', 'unknown')
    peers = sorted({r['name'] for r in rows if r.get('name') not in meshes
                    and r.get('type') == 'MESH'
                    and set(r.get('material_names', [])).intersection(materials)})
    return {'target_meshes': meshes, 'parent': parent,

def _location(program, target):
    # Only an exact literal include slot is a proven static mapping.
    # Dynamic names, helper expansion and Blender suffixes remain unknown.
    if not isinstance(target, str) or target == 'unknown':
        return 'unknown'
    try:
        tree = ast.parse(program)
    except (SyntaxError, TypeError):
        return 'unknown'
    lines = []
    for node in ast.walk(tree):
        if (isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute)
                and node.func.attr == 'include' and node.args
                and isinstance(node.args[0], ast.Constant)
                and node.args[0].value == target):
            lines.append(node.lineno)
    return 'program:line ' + str(lines[0]) if len(lines) == 1 else 'unknown'


def _dependencies(scene, target):
    rows = scene.get('structure', {}).get('objects', [])
    by_name = {r.get('name'): r for r in rows}
    regression = scene.get('regression', {})
    selection = regression.get('target_selection') or scene.get('structure', {}).get('target', {})
    meshes = selection.get('mesh_names', [])
    materials = sorted({m for name in meshes for m in by_name.get(name, {}).get('material_names', [])})
    parent = by_name.get(target, {}).get('parent', 'unknown')
    peers = sorted({r['name'] for r in rows if r.get('name') not in meshes
                    and r.get('type') == 'MESH'
                    and set(r.get('material_names', [])).intersection(materials)})
    return {'target_meshes': meshes, 'parent': parent,
            'observed_materials': materials, 'observed_shared_material_meshes': peers,
            'coverage': scene.get('structure', {}).get('evidence_projection', 'unknown'),
            'limitation': 'Projected rows cannot certify omitted dependencies; AABB relations are not true contact or collision.'}

    return ''


def _program(text):
    blocks = re.findall(r'```python\s*\n(.*?)```', text, re.S | re.I)
    return blocks[-1] if blocks else None

import math


def _failed(scene):
    return (scene.get('status') != 'complete' or
            (scene.get('metrics') or {}).get('valid') is not True)


def _score(scene):
    value = (scene.get('metrics') or {}).get('overall')
    if isinstance(value, (int, float)) and math.isfinite(value):
        return float(value)
    return float('inf')


def plan(packet):
    scenes = packet.get('development') or []
    failures = [s for s in scenes if _failed(s)]
    completed = [s for s in scenes if not _failed(s)]
    prior = packet.get('controlled_probes') or []
    used = max(len(prior), int(packet.get('executed_probe_count', 0) or 0))
    available = max(0, 2 - used)
    probes = []
    # Invalid artifacts cannot support camera/scale diagnostics. Address execution first.
    if not failures and completed and available:
        scene = min(completed, key=_score)
        functional = scene.get('functional') or {}
        regression = scene.get('regression') or {}
        broken_group = (functional.get('composite_pass') is False or
                        (regression.get('move') or {}).get('pass') is False or
                        (regression.get('material') or {}).get('pass') is False)
        kinds = ['isolate', 'object_scale'] if broken_group else ['camera_focal', 'object_scale']
        seen = {(p.get('scene_id'), p.get('kind')) for p in prior if isinstance(p, dict)}
        for kind in kinds:
            key = (scene.get('scene_id'), kind)
            if key[0] and key not in seen and len(probes) < available:
                probes.append({'scene_id': key[0], 'kind': kind})
    priority = ('First investigate missing artifacts, exceptions, timeouts and wasted calls; '
                'preserve successful-scene quality while improving reliability.' if failures else
                'Investigate the weakest measured valid reconstruction using available controlled '
                'diagnostics; separate camera error, object geometry and appearance.')
    instruction = (
        'Improve shared policy.solve(task,runtime) and callable meta.plan(packet). Inspect all supplied '

def _text(response):
    content = response.get('content') or ''
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        return '\n'.join(x.get('text', '') for x in content
                         if isinstance(x, dict) and x.get('type') == 'text')
    return ''
