def _failed(scene):

    if isinstance(value, (int, float)) and math.isfinite(value):
        return float(value)
    return float('inf')

Shared reconstruction procedure:

def _text(response):
    content = response.get('content') or ''
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        return '\n'.join(x.get('text', '') for x in content
                         if isinstance(x, dict) and x.get('type') == 'text')
    return ''

            return
        if not can_execute:
            return
        if text.upper() == 'DONE' and successful:
            return
        blocks = re.findall(r'```python\s*\n(.*?)```', text, re.S | re.I)
        if len(blocks) != 1:
            messages.append({'role': 'user', 'content':
                             'Return exactly one complete program in a python code block. '
                             'No valid build exists yet.' if not successful else
                             'Return exactly one complete program, DONE, or SELECT <successful step>.'})
            continue
        code = blocks[0]
        try:
            ast.parse(code)
        except SyntaxError as exc:
            messages.append({'role': 'user', 'content':
                             f'Python syntax failure before Blender execution: {exc}. '
                             'Return a corrected complete program. No execution was consumed.'})
            continue
        budget = runtime.remaining()
        if budget['executions'] <= 0 or budget['seconds'] <= 0:
            return
        attempts += 1
        result = runtime.execute(code)
        visible_success = bool(result.get('ok') and result.get('image'))
        if visible_success and isinstance(result.get('step'), int):
            successful[result['step']] = result['image']
        report = {k: v for k, v in result.items() if k != 'image'}
        if visible_success:
            instruction = (
                'Compare this render to the original. Identify the largest coherent mismatch, '
                'then return a complete revised program, DONE, or SELECT <successful step>. '
                'Preserve complete semantic groups and already matched features.')
        else:
            instruction = (
                'Execution/output failure: no usable successful render is available for this attempt. '
                'Do not describe or score an unseen image. Repair the error and audit all reliability '
                'requirements. For timeout, reduce expanded primitive count and surface detail sharply. '
                'Return a compact complete program if executions remain.')
        content = [{'type': 'text', 'text': json.dumps(report, default=str) + '\n' + instruction +
                    f'\nSuccessful steps: {list(successful)}. Remaining: {runtime.remaining()}'}]
        if visible_success:
            content.append(runtime.image(result['image']))
        messages.append({'role': 'user', 'content': content})

def _number(value):

        return True

        return True
    if 'attached_images' in scene and not scene['attached_images']:
        return True
    if 'program' in scene and not scene['program']:
        return True
    return False

    value = (scene.get('metrics') or {}).get('overall')
    return float(value) if _number(value) else None

    return probe.get('kind') or probe.get('probe') or (probe.get('result') or {}).get('probe')


def _events(scene):
    trajectory = scene.get('trajectory') or {}
    return trajectory.get('events') or [] if isinstance(trajectory, dict) else []


def _response_text(event):
    message = event.get('message') or {}
    content = message.get('content') or ''
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        return '\n'.join(part.get('text', '') for part in content if isinstance(part, dict))
    return ''


def _trajectory_audit(scene):
    events = [e for e in _events(scene) if isinstance(e, dict)]
    counts = {name: sum(e.get('event') == name for e in events)
              for name in ('model_request', 'model_response', 'codex_session',
                           'execute', 'execution_result', 'select')}
    pending = []
    unexecuted = 0
    for event in events:
        if event.get('event') == 'model_response':
            blocks = re.findall(r'```python\s*\n(.*?)```', _response_text(event), re.S | re.I)
            pending.extend(code.strip() for code in blocks)
        elif event.get('event') == 'execute':
            code = (event.get('code') or '').strip()
            if code in pending:
                index = pending.index(code)
                unexecuted += index
                pending = pending[index + 1:]
    results = [e for e in events if e.get('event') == 'execution_result']
    return {
        'scene_id': scene.get('scene_id'),
        'logged_counts': counts,
        'explicit_failed_execution_results': sum(e.get('ok') is not True for e in results),
        'execution_result_count_deficit': max(0, counts['execute'] - len(results)),
        'unexecuted_code_blocks_in_logged_responses': unexecuted + len(pending),
        'trajectory_available': bool(events),
        'limit': 'Logged events only; request/session/response counts are distinct. Missing event payloads do not identify a stalled stage.'
    }
