import bpy
from mathutils import Vector
s=bpy.context.scene
bpy.ops.object.camera_add(location=(0,-7,1.05))
c=bpy.context.object
c.rotation_euler=(Vector((0,0,0.7))-c.location).to_track_quat("-Z","Y").to_euler()
c.data.type="ORTHO"
c.data.ortho_scale=5.5
s.camera=c
s.render.filepath="/tmp/chair_side_check.png"
s.cycles.samples=16
bpy.ops.render.render(write_still=True)
