import bpy
from mathutils import Vector
dg=bpy.context.evaluated_depsgraph_get()
for o in [x for x in bpy.context.scene.objects if x.name.startswith("caster wheel")]:
 p=o.matrix_world.translation
 hit=bpy.context.scene.ray_cast(dg,Vector((p.x,p.y,.024)),Vector((0,0,-1)))
 print(o.name, "bottom",round(p.z-.075,5),"support",round(hit[1].z,5) if hit[0] else None)
